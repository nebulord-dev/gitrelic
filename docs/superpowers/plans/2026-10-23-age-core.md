# Age lens, core PR (RELIC-374) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every tracked file its real last change (not just files touched in the analysis window), and add orphaned files, woken files and dormant packages to `AgeMapReport`, as decided in `docs/atlas-views.md` › *Code Health › Age*.

**Architecture:** A new `getFileTouches` primitive in `utils/git.ts` reads the history *before* the window with `git log --name-only` (lighter than numstat). The runner maps its emails onto the window's merged identities and hands it to `analyzeAgeMap`, which merges both lists (deduped by hash), then derives last change, orphaned (cold, every human author inactive a year), woken (changed in the last year after a year untouched) and dormant packages (5+ source files, no source change in a year). If the older read fails, ages fall back to the window and `ageMap.source` says `'window'`. Stale Files (`deadCode`) needs no code change: it looks dates up in `ageMap.files`, which now holds every file.

**Tech Stack:** TypeScript 6, Vitest, execa (through `runGit`), pnpm + Turbo.

**Spec numbers to keep:** `COLD_DAYS = 365`, `DORMANT_MIN_SOURCE_FILES = 5`, sweeping commits (> `MAX_FIX_FILES` = 20 files) don't make anyone an author, non-human authors (bots, AI agents) never count as authors, the root (`packageOf(file) === ''`) is never a dormant package. Classic's fields (`status`, `thresholds`, `staleFiles`, `ancientFiles`, `summary`) keep their current meaning.

```ts
/** A commit's author, time and changed paths, without line counts. */
export interface FileTouch {
  hash: string;
  authorEmail: string;
  authorName: string;
  date: string; // ISO
  files: string[];
}

/**
 * The commits up to `until` (a git date, usually the window's `--since`), with
 * the paths each changed. Age reads it for the history before the window, so a
 * file untouched in the window still gets its real last change. `--name-only`
 * skips the line counts `getAllCommits` needs, which is most of a log's cost.
 */
export async function getFileTouches(
  repoPath: string,
  options: { until: string; ref?: string; timeoutMs?: number },
): Promise<FileTouch[]> {
  const args = [
    'log',
    '--format=COMMIT%x1F%H%x1F%aE%x1F%aN%x1F%aI',
    '--name-only',
    '--no-merges',
    `--until=${options.until}`,
  ];
  if (options.ref) args.push('--end-of-options', options.ref);
  const stdout = await runGit(args, {
    repoPath,
    timeoutMs: options.timeoutMs ?? GIT_HEAVY_TIMEOUT_MS,
    heavy: true,
  });
  return parseFileTouches(stdout);
}

export function parseFileTouches(raw: string): FileTouch[] {
  const touches: FileTouch[] = [];
  let current: FileTouch | null = null;
  for (const line of raw.split('\n')) {
    if (line.startsWith(HEADER_PREFIX)) {
      const [, hash, authorEmail, authorName, date] = line.split(FIELD_SEP);
      current = { hash, authorEmail, authorName, date, files: [] };
      touches.push(current);
    } else if (current && line) {
      current.files.push(line);
    }
  }
  return touches;
}
```

```ts
 * outside the window (Age's older history) onto the window's identities.

export function canonicalEmailMap(
  identities: IdentityAlias[],
): Map<string, string> {
  const map = new Map<string, string>();
  for (const id of identities)
    for (const alias of id.aliases)
      map.set(alias.email.toLowerCase(), id.canonicalEmail);
  return map;
}
```

```ts
import { packageOf } from './commit-timing.js';
import { MAX_FIX_FILES } from './fix-rate.js';
import type {
  AgeMapReport,
  AgeStatus,
  DormantPackage,
  FileAge,
  GitrelicReport,
} from '../types.js';
import { classifyAuthor } from '../utils/authorClassification.js';
import type { FileTouch, RawCommit } from '../utils/git.js';

/** A file unchanged this long is cold; a person without a commit this long is inactive. */
export const COLD_DAYS = 365;
/** A dormant package has at least this many source files, so a stray folder doesn't badge. */
export const DORMANT_MIN_SOURCE_FILES = 5;
const ORPHAN_AUTHORS_SHOWN = 3;
const DAY_MS = 86_400_000;

export interface AgeMapOptions {
  /**
   * The history before the window (`getFileTouches`), emails already mapped
   * to the window's identities. Undefined when the window is all of history;
   * null when it couldn't be read, so ages cover the window only.
   */
  older?: FileTouch[] | null;
  /** The report's non-source files: tests don't keep a package awake. */
  fileClasses?: GitrelicReport['fileClasses'];
}

interface Change {
  at: number;
  date: string;
}

/**
 * Ages every tracked file by its last change, from the window's commits and
 * the older history: ages count back from `asOf` (the analyzed commit), never
 * from today. Tiers (`status`) stay shares of `repoAgeDays` for classic.
 */
export function analyzeAgeMap(
  commits: RawCommit[],
  trackedFiles: string[],
  repoAgeDays: number,
  asOf: number,
  options: AgeMapOptions = {},
): AgeMapReport {
  const trackedSet = new Set(trackedFiles);
  const touches: FileTouch[] = [...commits];
  const inWindow = new Set(commits.map((c) => c.hash));
  // --since and --until overlap at the window's edge: keep each commit once.
  for (const t of options.older ?? []) if (!inWindow.has(t.hash)) touches.push(t);

  const changes = new Map<string, Change[]>();
  // Per file, its human authors' change counts (sweeping commits left out, as
  // in fix rate: a mass rename doesn't make you someone who knows the file).
  const authors = new Map<string, Map<string, number>>();
  const names = new Map<string, string>();
  const lastActive = new Map<string, number>();
  for (const t of touches) {
    const at = Date.parse(t.date);
    // Compare instants: ISO strings with different UTC offsets don't sort
    // chronologically as text.
    if (at > (lastActive.get(t.authorEmail) ?? -Infinity))
      lastActive.set(t.authorEmail, at);
    names.set(t.authorEmail, t.authorName);
    const counts =
      t.files.length <= MAX_FIX_FILES &&
      classifyAuthor(t.authorEmail) === 'human';
    for (const file of t.files) {
      if (!trackedSet.has(file)) continue;
      let list = changes.get(file);
      if (!list) changes.set(file, (list = []));
      list.push({ at, date: t.date });
      if (!counts) continue;
      let who = authors.get(file);
      if (!who) authors.set(file, (who = new Map()));
      who.set(t.authorEmail, (who.get(t.authorEmail) ?? 0) + 1);
    }
  }

  const thresholds = getAgeThresholds(repoAgeDays);
  const activeSince = asOf - COLD_DAYS * DAY_MS;
  const files: FileAge[] = [];
  for (const [file, list] of changes) {
    list.sort((a, b) => b.at - a.at);
    const ageInDays = Math.floor((asOf - list[0].at) / DAY_MS);
    const entry: FileAge = {
      file,
      lastCommitDate: list[0].date,
      ageInDays,
      status: getAgeStatus(ageInDays, thresholds),
    };
    if (ageInDays >= COLD_DAYS) {
      const who = authors.get(file);
      if (
        who &&
        [...who.keys()].every((e) => (lastActive.get(e) ?? -Infinity) <= activeSince)
      ) {
        entry.orphaned = true;
        entry.orphanedAuthors = [...who]
          .sort((a, b) => b[1] - a[1])
          .slice(0, ORPHAN_AUTHORS_SHOWN)
          .map(([email]) => names.get(email) ?? email);
      }
    } else {
      const gap = wokeAfter(list, activeSince);
      if (gap !== undefined) entry.wokeAfterDays = gap;
    }
    files.push(entry);
  }
  files.sort((a, b) => b.ageInDays - a.ageInDays);

  const staleFiles = files.filter((f) => f.status === 'stale');
  const ancientFiles = files.filter((f) => f.status === 'ancient');
  const ages = files.map((f) => f.ageInDays).sort((a, b) => a - b);
  const medianAgeDays = ages[Math.floor(ages.length / 2)] ?? 0;

  const summary =
    ancientFiles.length > 0
      ? `${ancientFiles.length} files haven't been touched in over ${thresholds.staleLimit} days — they may be dead weight or critical infrastructure nobody dares touch`
      : staleFiles.length > 0
        ? `${staleFiles.length} files are going stale (no commits in ${thresholds.agingLimit}+ days)`
        : 'The codebase is actively maintained across most files';

  return {
    files,
    staleFiles,
    ancientFiles,
    medianAgeDays,
    thresholds,
    source: options.older === null ? 'window' : 'history',
    dormantPackages: dormantPackages(files, options.fileClasses ?? {}),
    summary,
  };
}

/**
 * The longest-ago wake among the file's changes since `activeSince`: the first
 * gap of `COLD_DAYS` or more before one of them, newest first. `list` is
 * sorted newest first.
 */
function wokeAfter(list: Change[], activeSince: number): number | undefined {
  for (let i = 0; i + 1 < list.length && list[i].at > activeSince; i++) {
    const gap = list[i].at - list[i + 1].at;
    if (gap >= COLD_DAYS * DAY_MS) return Math.floor(gap / DAY_MS);
  }
  return undefined;
}

function dormantPackages(
  files: FileAge[],
  fileClasses: GitrelicReport['fileClasses'],
): DormantPackage[] {
  const byPackage = new Map<string, { sourceFiles: number; newest: number }>();
  for (const f of files) {
    if (fileClasses[f.file]) continue; // tests, fixtures, generated
    const pkg = packageOf(f.file);
    if (pkg === '') continue; // the root is the repo, never a package to archive
    const acc = byPackage.get(pkg) ?? { sourceFiles: 0, newest: Infinity };
    acc.sourceFiles++;
    acc.newest = Math.min(acc.newest, f.ageInDays);
    byPackage.set(pkg, acc);
  }
  return [...byPackage]
    .filter(([, p]) => p.sourceFiles >= DORMANT_MIN_SOURCE_FILES && p.newest >= COLD_DAYS)
    .map(([pkg, p]) => ({ package: pkg, sourceFiles: p.sourceFiles, newestAgeDays: p.newest }))
    .sort((a, b) => b.newestAgeDays - a.newestAgeDays);
}
```
